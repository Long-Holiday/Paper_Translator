cmake_minimum_required(VERSION 3.21)
project(PaperTranslator VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Pdf PdfWidgets Sql)
add_library(paper_core
    src/app_paths.cpp src/paper_repository.cpp src/translation_controller.cpp
    src/reader_widget.cpp src/settings_dialog.cpp src/main_window.cpp)
target_include_directories(paper_core PUBLIC src)
target_link_libraries(paper_core PUBLIC Qt6::Widgets Qt6::Pdf Qt6::PdfWidgets Qt6::Sql)
target_compile_definitions(paper_core PRIVATE PAPER_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
    target_compile_options(paper_core PUBLIC /utf-8 /W4)
else()
    target_compile_options(paper_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
qt_add_executable(PaperTranslator WIN32 src/main.cpp)
target_link_libraries(PaperTranslator PRIVATE paper_core)
install(TARGETS PaperTranslator RUNTIME DESTINATION .)
install(DIRECTORY engine DESTINATION . PATTERN "__pycache__" EXCLUDE)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(desktop_tests tests/desktop_tests.cpp)
    target_link_libraries(desktop_tests PRIVATE paper_core Qt6::Test)
    target_compile_definitions(desktop_tests PRIVATE PAPER_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME desktop_tests COMMAND desktop_tests)
    set_tests_properties(desktop_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
